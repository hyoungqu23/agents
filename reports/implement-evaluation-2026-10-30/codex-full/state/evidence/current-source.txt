## src/project-list.js
1: export function createProjectList({ loadProjects, onChange }) {
2:   let state = { status: 'all', projects: [], selected: null, loading: false, error: null };
3:   let request = 0;
4: 
5:   function update(changes) {
6:     state = { ...state, ...changes };
7:     onChange(state);
8:   }
9: 
10:   return {
11:     getState: () => state,
12:     select(project) {
13:       update({ selected: project });
14:     },
15:     async load(status = state.status) {
16:       const current = ++request;
17:       update({ status, projects: [], loading: true, error: null });
18:       try {
19:         const projects = await loadProjects(status);
20:         if (current !== request) return;
21:         update({ projects, loading: false, error: null });
22:       } catch (error) {
23:         if (current !== request) return;
24:         update({ projects: [], loading: false, error });
25:       }
26:     },
27:   };
28: }

## src/ui.js
1: export function selectField({ label, options, value, onChange }) {
2:   const wrapper = document.createElement('label');
3:   wrapper.append(document.createTextNode(`${label} `));
4:   const select = document.createElement('select');
5:   for (const [key, title] of options) select.add(new Option(title, key));
6:   select.value = value;
7:   select.addEventListener('change', () => onChange(select.value));
8:   wrapper.append(select);
9:   return wrapper;
10: }
11: 
12: export function renderRows(container, rows, selectedId, onSelect) {
13:   container.replaceChildren();
14:   for (const row of rows) {
15:     const button = document.createElement('button');
16:     button.className = 'project'; button.dataset.projectId = row.id;
17:     button.setAttribute('aria-pressed', String(selectedId === row.id));
18:     button.textContent = `${row.name} — ${row.status}`;
19:     button.addEventListener('click', () => onSelect(row));
20:     container.append(button);
21:   }
22: }
23: 
24: export function showProblem(container, retry) {
25:   const problem = document.createElement('div'); problem.className = 'problem';
26:   problem.setAttribute('role', 'alert'); problem.append('Could not load projects. ');
27:   const button = document.createElement('button'); button.textContent = 'Retry';
28:   button.addEventListener('click', retry); problem.append(button);
29:   container.replaceChildren(problem);
30: }

## src/api.js
1: export async function projects(status = 'all') {
2:   const response = await fetch(`/api/projects?status=${encodeURIComponent(status)}`);
3:   if (!response.ok) throw new Error('Could not load projects');
4:   return response.json();
5: }

8: const all = [tide, forest, north];

10: function deferred() {
11:   let resolve, reject;

16: test('filters fetch matching results and selection survives hidden and empty results', async () => {
17:   const calls = [];
18:   const list = createProjectList({
19:     loadProjects: async status => {
20:       calls.push(status);
21:       return status === 'all' ? all : all.filter(project => project.status === status);
22:     },
23:     onChange: () => {},
24:   });
